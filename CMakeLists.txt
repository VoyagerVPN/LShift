cmake_minimum_required(VERSION 3.25)
project(LShift VERSION 0.1.11 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /permissive- /utf-8 /EHsc)
add_compile_options("$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:/GL>")
add_link_options("$<$<CONFIG:Release>:/LTCG>" "$<$<CONFIG:Release>:/OPT:REF,ICF>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(lshift_core src/core.cpp src/recent_input.cpp)
target_include_directories(lshift_core PUBLIC src)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE lshift_core)
add_test(NAME core COMMAND core_tests)
add_executable(recent_input_tests tests/recent_input_tests.cpp)
target_link_libraries(recent_input_tests PRIVATE lshift_core)
add_test(NAME recent_input COMMAND recent_input_tests)
option(LSHIFT_BUILD_APP "Build the Windows tray application" ON)
option(LSHIFT_DIAGNOSTICS "Include development logging in the application" OFF)
set(LSHIFT_APP_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist" CACHE PATH "Application output directory")
configure_file(resources/README.txt "${LSHIFT_APP_OUTPUT_DIRECTORY}/README.txt" COPYONLY)
set_source_files_properties(resources/app.rc PROPERTIES OBJECT_DEPENDS
  "${CMAKE_SOURCE_DIR}/resources/app.ico;${CMAKE_SOURCE_DIR}/LICENSE;${CMAKE_SOURCE_DIR}/NOTICE")
if(LSHIFT_BUILD_APP)
  add_library(lshift_windows src/shell.cpp src/text_service.cpp src/text_context.cpp src/clipboard.cpp)
  target_compile_definitions(lshift_windows PUBLIC LSHIFT_DIAGNOSTICS=$<BOOL:${LSHIFT_DIAGNOSTICS}>)
  target_link_libraries(lshift_windows PUBLIC lshift_core user32 shell32 gdi32 advapi32 ole32 oleaut32 uuid comctl32 uiautomationcore)
  add_executable(LShift WIN32 src/main.cpp resources/app.rc)
  target_link_libraries(LShift PRIVATE lshift_windows)
  add_library(lshift_diagnostics src/diagnostics.cpp)
  target_include_directories(lshift_diagnostics PUBLIC src)
  if(LSHIFT_DIAGNOSTICS)
    target_link_libraries(LShift PRIVATE lshift_diagnostics)
  endif()
  add_executable(shell_tests tests/shell_tests.cpp)
  target_link_libraries(shell_tests PRIVATE lshift_windows)
  add_test(NAME settings COMMAND shell_tests)
  set_tests_properties(settings PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(native_tests tests/native_tests.cpp)
  target_link_libraries(native_tests PRIVATE lshift_windows)
  add_test(NAME native_text COMMAND native_tests)
  set_tests_properties(native_text PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
  add_executable(ui_tests tests/ui_tests.cpp resources/app.rc)
  target_link_libraries(ui_tests PRIVATE lshift_windows windowscodecs)
  target_link_options(ui_tests PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_SOURCE_DIR}/resources/app.manifest)
  add_executable(tray_tests tests/tray_tests.cpp)
  target_link_libraries(tray_tests PRIVATE lshift_windows)
  target_link_options(tray_tests PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_SOURCE_DIR}/resources/app.manifest)
  add_executable(automation_probe tests/automation_probe.cpp)
  target_link_libraries(automation_probe PRIVATE lshift_windows)
  add_executable(clipboard_tests tests/clipboard_tests.cpp)
  target_link_libraries(clipboard_tests PRIVATE lshift_windows)
  add_test(NAME clipboard COMMAND clipboard_tests)
  set_tests_properties(clipboard PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
  add_executable(diagnostic_tests tests/diagnostic_tests.cpp)
  target_link_libraries(diagnostic_tests PRIVATE lshift_diagnostics)
  add_test(NAME diagnostics COMMAND diagnostic_tests)
  set_tests_properties(diagnostics PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(text_context_tests tests/text_context_tests.cpp)
  target_link_libraries(text_context_tests PRIVATE lshift_windows)
  add_test(NAME text_context COMMAND text_context_tests)
  set_target_properties(LShift PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LSHIFT_APP_OUTPUT_DIRECTORY}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${LSHIFT_APP_OUTPUT_DIRECTORY}")
  target_link_options(LShift PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_SOURCE_DIR}/resources/app.manifest)
endif()
